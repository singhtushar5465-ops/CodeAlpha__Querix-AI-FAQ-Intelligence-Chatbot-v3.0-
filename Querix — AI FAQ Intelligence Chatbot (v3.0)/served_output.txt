201:   function renderVoiceOrb() {
202:     ctx.clearRect(0, 0, canvas.width, canvas.height);
203:     const centerX = canvas.width / 2;
204:     const centerY = canvas.height / 2;
205:     orbAngle += 0.03;
206: 
207:     let baseRadius = 46;
208:     let primaryColor = 'rgba(108, 92, 231, ';
209:     let secondaryColor = 'rgba(0, 212, 255, ';
210: 
211:     if (state.voiceState === 'LISTENING') {
212:       baseRadius = 54 + Math.sin(orbAngle * 4) * 8;
213:       primaryColor = 'rgba(245, 166, 35, ';
214:       secondaryColor = 'rgba(0, 212, 255, ';
215:     } else if (state.voiceState === 'THINKING') {
216:       baseRadius = 50 + Math.cos(orbAngle * 6) * 6;
217:       primaryColor = 'rgba(255, 107, 157, ';
218:       secondaryColor = 'rgba(108, 92, 231, ';
219:     } else if (state.voiceState === 'SPEAKING') {
220:       baseRadius = 58 + Math.sin(orbAngle * 5) * 10;
221:       primaryColor = 'rgba(0, 212, 255, ';
222:       secondaryColor = 'rgba(46, 204, 113, ';
223:     } else {
224:       // IDLE breathing
225:       baseRadius = 46 + Math.sin(orbAngle * 1.5) * 4;
226:     }
227: 
228:     // Outer Halo
229:     const outerGrad = ctx.createRadialGradient(centerX, centerY, baseRadius * 0.4, centerX, centerY, baseRadius * 1.8);
230:     outerGrad.addColorStop(0, primaryColor + '0.5)');
231:     outerGrad.addColorStop(0.6, secondaryColor + '0.2)');
232:     outerGrad.addColorStop(1, 'transparent');
233:     ctx.fillStyle = outerGrad;
234:     ctx.beginPath();
235:     ctx.arc(centerX, centerY, baseRadius * 1.8, 0, Math.PI * 2);
236:     ctx.fill();
237: 
238:     // Multi-Layer Harmonic Wave Rings
239:     const layers = state.voiceState === 'SPEAKING' ? 6 : 4;
240:     for (let i = 0; i < layers; i++) {
241:       ctx.beginPath();
242:       const phase = orbAngle * (i % 2 === 0 ? 1 : -1) + (i * Math.PI) / 3;
243:       const waveCount = 8;
244:       for (let a = 0; a <= Math.PI * 2; a += 0.05) {
245:         const distortion = Math.sin(a * waveCount + phase) * (state.voiceState === 'IDLE' ? 4 : 14);
246:         const r = baseRadius + distortion;
247:         const x = centerX + Math.cos(a) * r;
248:         const y = centerY + Math.sin(a) * r;
249:         if (a === 0) ctx.moveTo(x, y);
250:         else ctx.lineTo(x, y);
251:       }
252:       ctx.closePath();
253:       ctx.strokeStyle = i % 2 === 0 ? primaryColor + '0.8)' : secondaryColor + '0.8)';
254:       ctx.lineWidth = 2.2;
255:       ctx.stroke();
256:     }
257: 
258:     // Core Radiant Sphere
259:     const coreGrad = ctx.createRadialGradient(centerX - 15, centerY - 15, 5, centerX, centerY, baseRadius);
260:     coreGrad.addColorStop(0, '#FFFFFF');
261:     coreGrad.addColorStop(0.4, primaryColor + '0.9)');
262:     coreGrad.addColorStop(1, secondaryColor + '1)');
263:     ctx.fillStyle = coreGrad;
264:     ctx.beginPath();
265:     ctx.arc(centerX, centerY, baseRadius * 0.75, 0, Math.PI * 2);
266:     ctx.fill();
267: 
268:     animationFrameId = requestAnimationFrame(renderVoiceOrb);
269:   }

324:         const trans = event.results[i][0].transcript;

326:           finalTranscript += trans;

336:         chatTextarea.value = display;
337:         hasSubmittedCurrentSpeech = false;
338: 
339:         // Auto-submit after 850ms pause in speech

342:           if (accumulatedSpeech.trim() && !hasSubmittedCurrentSpeech && state.voiceState === 'LISTENING') {

346:       }
347:     };

483:   function speakText(text) {
484:     if (!synth || !state.autoReadAnswers) return;
485:     try {
486:       synth.cancel();
487:       if (synth.paused) synth.resume();
488: 
489:       const cleanSpeech = text.replace(/[*_#`]/g, '').slice(0, 360);
490:       const utterance = new SpeechSynthesisUtterance(cleanSpeech);
491:       utterance.rate = state.speechSpeed;
492:       utterance.lang = state.speechLang === 'hinglish' ? 'en-IN' : state.speechLang;
493: 
494:       // Select voice if available
495:       const voices = synth.getVoices();
496:       if (voices.length > 0) {
497:         const bestVoice = voices.find(v => v.lang.includes('IN') || v.lang.includes('hi')) ||
498:                           voices.find(v => v.lang.includes('en') && v.name.includes('Google')) ||
499:                           voices.find(v => v.lang.includes('en')) ||
500:                           voices[0];
501:         if (bestVoice) utterance.voice = bestVoice;
502:       }
503: 
504:       utterance.onstart = () => {
505:         setVoiceState('SPEAKING', 'Jarvix is answering aloud… (Tap to interrupt)');
506:         startAudioMeterAnimation();
507:       };
508: 
509:       utterance.onend = () => {
510:         setVoiceState('IDLE', 'Tap orb or speak to ask another question');
511:         stopAudioMeterAnimation();
512:         // Hands-free follow up
513:         if (state.isVoiceModalOpen) {
514:           setTimeout(() => {
515:             if (state.isVoiceModalOpen && state.voiceState === 'IDLE') {
516:               startListening();
517:             }
518:           }, 1200);
519:         }
520:       };
521: 
522:       utterance.onerror = () => {
523:         setVoiceState('IDLE', 'Finished speaking.');
524:         stopAudioMeterAnimation();
525:       };
526: 
527:       synth.speak(utterance);
528:     } catch (err) {
529:       console.warn('Speech synthesis error:', err);
530:       setVoiceState('IDLE', 'Answer displayed.');
531:       stopAudioMeterAnimation();
532:     }

687:   async function handleUserQuery(query, isVoice = false) {
688:     if (!query) return;
689: 
690:     // Hide empty state hero
691:     if (emptyState.style.display !== 'none') {
692:       emptyState.style.display = 'none';
693:     }
694: 
695:     // Append User Bubble
696:     appendUserBubble(query);
697: 
698:     // Show Typing Indicator
699:     typingIndicator.style.display = 'flex';
700:     scrollToBottom();
701: 
702:     if (state.isVoiceModalOpen) {
703:       setVoiceState('THINKING', 'Jarvix is consulting knowledge base…');
704:     }
705: 
706:     try {
707:       const response = await fetch(`${API_BASE}/api/chat`, {
708:         method: 'POST',
709:         headers: { 'Content-Type': 'application/json' },
710:         body: JSON.stringify({
711:           query: query,
712:           topic: state.currentTopic,
713:           is_voice: isVoice,
714:           session_id: state.sessionId,
715:           guest_id: state.guestId
716:         })
717:       });
718: 
719:       const data = await response.json();
720:       typingIndicator.style.display = 'none';
721: 
722:       // Soft Block / Guest limit check
723:       if (data.blocked) {
724:         if (data.reason === 'guest_limit') {
725:           appendBotBubble({
726:             matched: false,
727:             answer: data.message,
728:             confidence: 0,
729:             contact_support: false,
730:             isAlert: true
731:           });
732:           openAuthModal(true);
733:         } else if (data.reason === 'token_limit') {
734:           appendBotBubble({
735:             matched: false,
736:             answer: data.message,
737:             confidence: 0,
738:             contact_support: false,
739:             isAlert: true
740:           });
741:           openPricingModal();
742:         }
743:         return;
744:       }
745: 
746:       // Update token metrics & usage
747:       updateTokenUsageMetrics(data);
748: 
749:       // Append Bot Bubble with answer & confidence
750:       appendBotBubble(data, isVoice);
751: 
752:       // Voice Modal integration
753:       if (state.isVoiceModalOpen) {
754:         voiceAnswerBox.style.display = 'block';
755:         voiceMatchBadge.textContent = `${data.confidence}% Match`;
756:         voiceTopicLabel.textContent = data.topic || 'General';
757:         voiceAnswerText.textContent = data.answer;
758:         if (state.autoReadAnswers) {
759:           speakText(data.answer);
760:         } else {
761:           setVoiceState('IDLE', 'Answer received. Tap orb to speak again.');
762:         }
763:       }
764: 
765:     } catch (err) {
766:       console.error('Chat error:', err);
767:       typingIndicator.style.display = 'none';
768:       appendBotBubble({
769:         matched: false,
770:         answer: 'Querix encountered a network connection anomaly. Please check backend server status.',
771:         confidence: 0,
772:         contact_support: true
773:       });
774:     }
775: 
776:     scrollToBottom();
777:   }
